# 开发指南 - 自定义测试算子

---

## 1. 概述

TestPipe框架支持自定义测试算子,让你可以扩展框架功能,实现特定的测试逻辑。

---

## 2. 测试算子基础

### 2.1 TestOp基类

所有测试算子必须继承`TestOp`基类:

```python
from testpipe.core import TestOp
from typing import Dict, Any

class MyTestOp(TestOp):
    """自定义测试算子"""
    
    # 算子元信息
    op_type = "MyTest"              # 算子类型(必需)
    op_category = "custom"          # 算子分类(可选)
    op_version = "1.0"              # 算子版本(可选)
    
    def execute(self, context) -> Dict[str, Any]:
        """执行算子逻辑(必需实现)"""
        # 获取输入
        input_value = context.get(self.inputs[0])
        
        # 执行逻辑
        result = self.process(input_value)
        
        # 返回输出
        return {
            self.outputs[0]: result
        }
```

### 2.2 算子注册

使用`@register_op`装饰器自动注册:

```python
from testpipe.core import register_op

@register_op
class MyTestOp(TestOp):
    op_type = "MyTest"
    
    def execute(self, context):
        return {"result": "ok"}
```

## 3. 完整示例

### 3.1 简单算子 - Echo

```python
from testpipe.core import TestOp, register_op
from typing import Dict, Any

@register_op
class EchoOp(TestOp):
    """回显算子 - 返回输入内容"""
    
    op_type = "Echo"
    op_category = "utility"
    
    def execute(self, context) -> Dict[str, Any]:
        """执行回显"""
        # 获取输入
        input_data = context.get(self.inputs[0])
        
        # 直接返回
        return {
            self.outputs[0]: input_data
        }
```

### 3.2 复杂算子 - ATCCompile

```python
from testpipe.core import TestOp, register_op, OpExecutionError
from typing import Dict, Any
import subprocess

@register_op
class ATCCompileOp(TestOp):
    """ATC模型编译算子"""
    
    op_type = "ATCCompile"
    op_category = "compile"
    
    def validate(self) -> bool:
        """验证配置"""
        # 检查必需属性
        if 'framework' not in self.attributes:
            raise OpExecutionError(self.name, "缺少framework属性")
        
        # 检查ATC工具是否可用
        try:
            subprocess.run(['which', 'atc'], check=True, capture_output=True)
        except subprocess.CalledProcessError:
            raise OpExecutionError(self.name, "ATC工具不可用")
        
        return True
    
    def execute(self, context) -> Dict[str, Any]:
        """执行ATC编译"""
        # 获取输入
        model_path = context.get('model_path')
        soc_version = context.get('soc_version')
        
        # 获取配置
        framework = self.get_attribute('framework', 5)
        output_type = self.get_attribute('output_type', 'FP16')
        timeout = self.get_timeout()
        
        # 构建命令
        output_path = f"{context.work_dir}/model.om"
        cmd = [
            'atc',
            f'--model={model_path}',
            f'--framework={framework}',
            f'--output={output_path}',
            f'--soc_version={soc_version}',
            f'--output_type={output_type}'
        ]
        
        # 执行编译
        self.logger.info(f"执行ATC编译: {' '.join(cmd)}")
        try:
            result = subprocess.run(
                cmd,
                timeout=timeout,
                capture_output=True,
                text=True,
                check=True
            )
            self.logger.info(f"编译成功: {output_path}")
        except subprocess.TimeoutExpired:
            raise OpExecutionError(self.name, f"编译超时({timeout}秒)")
        except subprocess.CalledProcessError as e:
            raise OpExecutionError(self.name, f"编译失败: {e.stderr}")
        
        # 返回输出
        return {
            'om_path': output_path
        }
    
    def teardown(self, context):
        """清理临时文件"""
        # 可选: 清理中间文件
        pass
```

## 4. 使用统一API

### 4.1 SSH操作

```python
def execute(self, context):
    # 连接SSH
    self.api.ssh.connect(
        host="192.168.1.100",
        port=22,
        user="root"
    )
    
    # 执行命令
    output = self.api.ssh.execute("ls -la")
    
    # 断开连接
    self.api.ssh.disconnect()
    
    return {"output": output}
```

```python
def execute(self, context):
```

## 5. 错误处理

### 5.1 抛出异常

```python
from testpipe.core import OpExecutionError, OpValidationError, OpTimeoutError

def execute(self, context):
    # 验证错误
    if not self.validate():
        raise OpValidationError(self.name, "配置验证失败")
    
    # 执行错误
    try:
        result = self.do_work()
    except Exception as e:
        raise OpExecutionError(self.name, f"执行失败: {str(e)}", cause=e)
    
    # 超时错误
    if elapsed_time > timeout:
        raise OpTimeoutError(self.name, timeout)
    
    return {"result": result}
```

## 6. 算子生命周期

```python
@register_op
class MyOp(TestOp):
    op_type = "MyOp"
    
    def validate(self) -> bool:
        """1. 验证配置(execute之前)"""
        return True
    
    def setup(self, context):
        """2. 准备工作(execute之前)"""
        self.temp_dir = context.work_dir / "temp"
        self.temp_dir.mkdir(exist_ok=True)
    
    def execute(self, context) -> Dict[str, Any]:
        """3. 执行逻辑"""
        return {"result": "ok"}
    
    def teardown(self, context):
        """4. 清理工作(execute之后,无论成功失败)"""
        if self.temp_dir.exists():
            shutil.rmtree(self.temp_dir)
```

## 7. 最佳实践

### 7.1 命名规范

- 算子类名: `XxxOp` (如`ATCCompileOp`)
- 算子类型: `Xxx` (如`ATCCompile`)
- 文件名: `xxx_op.py` (如`atc_compile_op.py`)

### 7.2 输入输出

- 使用清晰的输入输出名称
- 在Pipeline JSON中明确声明
- 通过`context.get()`获取输入
- 返回字典作为输出

### 7.3 配置属性

- 使用`self.get_attribute()`获取属性
- 提供合理的默认值
- 在`validate()`中检查必需属性

### 7.4 日志记录

```python
def execute(self, context):
    self.logger.info("开始执行")
    self.logger.debug(f"输入参数: {self.inputs}")
    
    try:
        result = self.do_work()
        self.logger.info("执行成功")
    except Exception as e:
        self.logger.error(f"执行失败: {e}")
        raise
    
    return result
```

### 7.5 超时处理

```python
def execute(self, context):
    timeout = self.get_timeout()  # 从attributes获取
    
    # 使用timeout参数
    result = subprocess.run(
        cmd,
        timeout=timeout,
        capture_output=True
    )
```

## 8. 测试算子

### 8.1 单元测试

```python
# tests/ops/test_my_op.py
import pytest
from testpipe.core import ExecutionContext
from testpipe.ops import MyOp

def test_my_op_basic():
    """测试基本功能"""
    op = MyOp(name="test_op", attributes={})
    op.inputs = ["input_data"]
    op.outputs = ["output_data"]
    
    context = ExecutionContext(work_dir="/tmp")
    context.set("input_data", "test")
    
    result = op.execute(context)
    
    assert result["output_data"] == "test"

def test_my_op_validation():
    """测试验证"""
    op = MyOp(name="test_op", attributes={})
    
    with pytest.raises(OpValidationError):
        op.validate()
```

## 10. 发布算子

### 10.1 作为插件发布

```python
# my_plugin/setup.py
from setuptools import setup

setup(
    name='testpipe-my-plugin',
    version='1.0.0',
    packages=['my_plugin'],
    install_requires=['testpipe'],
    entry_points={
        'testpipe.ops': [
            'MyOp = my_plugin.my_op:MyOp',
        ]
    }
)
```

### 10.2 安装插件

```bash
pip install testpipe-my-plugin
```

算子会自动注册到TestPipe。
