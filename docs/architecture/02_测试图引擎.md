## 1. 需求描述

### 1.1 业务需求

TestPipe框架需要一个Pipeline引擎来管理测试流程:
- 将测试流程抽象为有向无环图(DAG)
- 自动分析节点依赖关系
- 按拓扑顺序执行测试算子
- 管理节点间的数据流
- 提供执行上下文存储中间产物

### 1.2 用户场景

**场景1: 定义测试流程**
```python
# 创建测试图
graph = TestGraph(name="ATC_E2E_Test")

# 添加节点
graph.add_node(env_check_op)
graph.add_node(atc_compile_op)

# 自动分析依赖关系(基于输入输出名称)
graph.build_edges()

# 执行测试
executor = TestEngine(graph)
result = executor.execute(inputs={"model_source": "..."})
```

**场景2: 查看执行顺序**
```python
# 获取拓扑排序后的执行顺序
order = graph.get_execution_order()
print(order)  # [env_check, model_prep, atc_compile, ...]
```

## 2. 设计方案

### 2.1 核心类设计

#### TestGraph - 测试图

```python
from typing import Dict, List, Set, Optional
from .test_op import TestOp

class TestGraph:
    """
    测试图 - 管理测试算子和数据流
    
    测试图是一个有向无环图(DAG),节点是测试算子,边是数据流
    """
    
    def __init__(self, name: str, description: str = ""):
        """
        初始化测试图
        
        Args:
            name: 图名称
            description: 图描述
        """
        self.name = name
        self.description = description
        
        # 节点管理
        self.nodes: Dict[str, TestOp] = {}  # node_name -> TestOp
        
        # 边管理(数据流)
        self.edges: Dict[str, List[str]] = {}  # output_name -> [consumer_node_names]
        
        # Pipeline输入输出
        self.pipeline_inputs: List[str] = []   # Pipeline级别的输入
        self.pipeline_outputs: List[str] = []  # Pipeline级别的输出
        
        # 缓存
        self._execution_order: Optional[List[TestOp]] = None
        self._is_validated: bool = False
    
    # ============================================
    # 节点管理
    # ============================================
    def add_node(self, node: TestOp):
        """
        添加节点
        
        Args:
            node: 测试算子实例
            
        Raises:
            ValueError: 节点名称重复
        """
        if node.name in self.nodes:
            raise ValueError(f"节点名称重复: {node.name}")
        
        self.nodes[node.name] = node
        self._invalidate_cache()
    
    def get_node(self, name: str) -> TestOp:
        """获取节点"""
        if name not in self.nodes:
            raise KeyError(f"节点不存在: {name}")
        return self.nodes[name]
    
    def remove_node(self, name: str):
        """删除节点"""
        if name in self.nodes:
            del self.nodes[name]
            self._invalidate_cache()
    
    # ============================================
    # 边管理(数据流)
    # ============================================
    def build_edges(self):
        """
        自动构建边(数据流)
        
        根据节点的输入输出名称自动建立连接:
        - 如果节点A的输出名称 == 节点B的输入名称,则建立边 A -> B
        """
        self.edges.clear()
        
        # 收集所有输出
        all_outputs: Dict[str, str] = {}  # output_name -> producer_node_name
        for node in self.nodes.values():
            for output in node.outputs:
                if output in all_outputs:
                    raise ValueError(
                        f"输出名称冲突: {output} "
                        f"(节点 {all_outputs[output]} 和 {node.name})"
                    )
                all_outputs[output] = node.name
        
        # 建立边
        for node in self.nodes.values():
            for input_name in node.inputs:
                if input_name in all_outputs:
                    # 找到生产者节点
                    producer = all_outputs[input_name]
                    
                    # 添加边: producer -> consumer
                    if input_name not in self.edges:
                        self.edges[input_name] = []
                    self.edges[input_name].append(node.name)
        
        self._invalidate_cache()
    
    def add_edge(self, output_name: str, consumer_node: str):
        """
        手动添加边
        
        Args:
            output_name: 输出名称
            consumer_node: 消费者节点名称
        """
        if consumer_node not in self.nodes:
            raise ValueError(f"消费者节点不存在: {consumer_node}")
        
        if output_name not in self.edges:
            self.edges[output_name] = []
        
        if consumer_node not in self.edges[output_name]:
            self.edges[output_name].append(consumer_node)
            self._invalidate_cache()
    
    # ============================================
    # 拓扑排序
    # ============================================
    def get_execution_order(self) -> List[TestOp]:
        """
        获取执行顺序(拓扑排序)
        
        Returns:
            按执行顺序排列的节点列表
            
        Raises:
            GraphCycleError: 图中存在环
        """
        if self._execution_order is not None:
            return self._execution_order
        
        # 构建依赖关系图
        dependencies: Dict[str, Set[str]] = {}  # node -> {依赖的节点}
        for node in self.nodes.values():
            dependencies[node.name] = set()
        
        # 分析依赖
        for node in self.nodes.values():
            for input_name in node.inputs:
                # 找到生产这个输入的节点
                for producer_node in self.nodes.values():
                    if input_name in producer_node.outputs:
                        dependencies[node.name].add(producer_node.name)
        
        # Kahn算法进行拓扑排序
        in_degree = {name: len(deps) for name, deps in dependencies.items()}
        queue = [name for name, degree in in_degree.items() if degree == 0]
        result = []
        
        while queue:
            # 取出入度为0的节点
            current = queue.pop(0)
            result.append(self.nodes[current])
            
            # 更新依赖此节点的其他节点的入度
            for node_name, deps in dependencies.items():
                if current in deps:
                    in_degree[node_name] -= 1
                    if in_degree[node_name] == 0:
                        queue.append(node_name)
        
        # 检查是否有环
        if len(result) != len(self.nodes):
            raise GraphCycleError("测试图中存在环,无法执行")
        
        self._execution_order = result
        return result
    
    # ============================================
    # 验证
    # ============================================
    def validate(self) -> bool:
        """
        验证图的正确性
        
        检查:
        1. 无环
        2. 所有输入都有来源(除了Pipeline输入)
        3. Pipeline输出都有生产者
        4. 节点配置正确
        
        Returns:
            True表示验证通过
            
        Raises:
            GraphValidationError: 验证失败
        """
        if self._is_validated:
            return True
        
        # 1. 检查无环(通过拓扑排序)
        try:
            self.get_execution_order()
        except GraphCycleError as e:
            raise GraphValidationError(f"图验证失败: {e}")
        
        # 2. 检查所有输入都有来源
        all_outputs = set()
        for node in self.nodes.values():
            all_outputs.update(node.outputs)
        
        for node in self.nodes.values():
            for input_name in node.inputs:
                if input_name not in all_outputs and input_name not in self.pipeline_inputs:
                    raise GraphValidationError(
                        f"节点 {node.name} 的输入 {input_name} 没有来源"
                    )
        
        # 3. 检查Pipeline输出都有生产者
        for output_name in self.pipeline_outputs:
            if output_name not in all_outputs:
                raise GraphValidationError(
                    f"Pipeline输出 {output_name} 没有生产者"
                )
        
        # 4. 验证每个节点的配置
        for node in self.nodes.values():
            if not node.validate():
                raise GraphValidationError(
                    f"节点 {node.name} 配置验证失败"
                )
        
        self._is_validated = True
        return True
    
    # ============================================
    # 辅助方法
    # ============================================
    def _invalidate_cache(self):
        """清除缓存"""
        self._execution_order = None
        self._is_validated = False
    
    def get_node_dependencies(self, node_name: str) -> List[str]:
        """获取节点的依赖节点"""
        node = self.get_node(node_name)
        dependencies = []
        
        for input_name in node.inputs:
            for producer in self.nodes.values():
                if input_name in producer.outputs:
                    dependencies.append(producer.name)
        
        return dependencies
    
    def get_node_consumers(self, node_name: str) -> List[str]:
        """获取节点的消费者节点"""
        node = self.get_node(node_name)
        consumers = []
        
        for output_name in node.outputs:
            if output_name in self.edges:
                consumers.extend(self.edges[output_name])
        
        return list(set(consumers))
    
    def __repr__(self) -> str:
        return f"<TestGraph(name={self.name}, nodes={len(self.nodes)})>"
```

### 2.2 执行上下文

#### ExecutionContext - 执行上下文

```python
from pathlib import Path
from typing import Dict, Any

class ExecutionContext:
    """
    执行上下文 - 存储测试执行过程中的数据
    
    上下文在整个测试流程中传递,用于:
    - 存储节点间传递的中间产物
    - 提供工作目录
    - 存储元数据和共享状态
    """
    
    def __init__(self, work_dir: Path):
        """
        初始化执行上下文
        
        Args:
            work_dir: 工作目录
        """
        self.work_dir = work_dir
        self.data: Dict[str, Any] = {}       # 中间产物
        self.metadata: Dict[str, Any] = {}   # 元数据
        
        # 确保工作目录存在
        self.work_dir.mkdir(parents=True, exist_ok=True)
    
    # ============================================
    # 数据访问
    # ============================================
    def set(self, key: str, value: Any):
        """
        设置数据
        
        Args:
            key: 数据键(通常是输出名称)
            value: 数据值
        """
        self.data[key] = value
    
    def get(self, key: str, default: Any = None) -> Any:
        """
        获取数据
        
        Args:
            key: 数据键
            default: 默认值
            
        Returns:
            数据值,如果不存在返回default
        """
        return self.data.get(key, default)
    
    def has(self, key: str) -> bool:
        """检查数据是否存在"""
        return key in self.data
    
    def require(self, key: str) -> Any:
        """
        获取必需的数据
        
        Args:
            key: 数据键
            
        Returns:
            数据值
            
        Raises:
            KeyError: 数据不存在
        """
        if key not in self.data:
            raise KeyError(f"上下文中缺少必需的数据: {key}")
        return self.data[key]
    
    # ============================================
    # 元数据访问
    # ============================================
    def set_metadata(self, key: str, value: Any):
        """设置元数据"""
        self.metadata[key] = value
    
    def get_metadata(self, key: str, default: Any = None) -> Any:
        """获取元数据"""
        return self.metadata.get(key, default)
    
    # ============================================
    # 工作目录管理
    # ============================================
    def get_node_work_dir(self, node_name: str) -> Path:
        """
        获取节点的工作目录
        
        Args:
            node_name: 节点名称
            
        Returns:
            节点工作目录路径
        """
        node_dir = self.work_dir / node_name
        node_dir.mkdir(parents=True, exist_ok=True)
        return node_dir
    
    def __repr__(self) -> str:
        return f"<ExecutionContext(work_dir={self.work_dir}, data_keys={len(self.data)})>"
```

### 2.3 测试引擎

#### TestEngine - 测试执行引擎

```python
from typing import Dict, Any
from pathlib import Path
import time

class TestEngine:
    """
    测试执行引擎
    
    负责执行测试图,管理执行流程
    """
    
    def __init__(self, graph: TestGraph, work_dir: Path = None):
        """
        初始化测试引擎
        
        Args:
            graph: 测试图
            work_dir: 工作目录,默认为当前目录下的.tge_work
        """
        self.graph = graph
        self.work_dir = work_dir or Path.cwd() / ".tge_work"
        self.api = OpAPI()  # 统一操作API
    
    def execute(self, inputs: Dict[str, Any]) -> Dict[str, Any]:
        """
        执行测试图
        
        Args:
            inputs: Pipeline输入参数
            
        Returns:
            Pipeline输出结果
            
        Raises:
            GraphValidationError: 图验证失败
            OpExecutionError: 节点执行失败
        """
        # 1. 验证图
        self.graph.validate()
        
        # 2. 创建执行上下文
        context = ExecutionContext(work_dir=self.work_dir)
        
        # 3. 设置Pipeline输入
        for key, value in inputs.items():
            if key not in self.graph.pipeline_inputs:
                raise ValueError(f"未知的Pipeline输入: {key}")
            context.set(key, value)
        
        # 4. 获取执行顺序
        execution_order = self.graph.get_execution_order()
        
        print(f"\n{'='*60}")
        print(f"开始执行测试: {self.graph.name}")
        print(f"{'='*60}")
        print(f"节点数量: {len(execution_order)}")
        print(f"执行顺序: {' → '.join([n.name for n in execution_order])}")
        print(f"{'='*60}\n")
        
        # 5. 按顺序执行节点
        for i, node in enumerate(execution_order, 1):
            print(f"[{i}/{len(execution_order)}] 执行节点: {node.name} ({node.op_type})")
            
            try:
                # 注入API
                node.api = self.api
                
                # 执行setup
                node.setup(context)
                
                # 执行节点
                start_time = time.time()
                outputs = node.execute(context)
                elapsed = time.time() - start_time
                
                # 保存输出到上下文
                for output_name, output_value in outputs.items():
                    context.set(output_name, output_value)
                
                print(f"  ✓ 完成 (耗时: {elapsed:.2f}秒)")
                print(f"  输出: {list(outputs.keys())}")
                
                # 执行teardown
                node.teardown(context)
                
            except Exception as e:
                print(f"  ✗ 失败: {e}")
                raise OpExecutionError(
                    op_name=node.name,
                    message=str(e),
                    cause=e
                )
        
        # 6. 收集Pipeline输出
        results = {}
        for output_name in self.graph.pipeline_outputs:
            if not context.has(output_name):
                raise RuntimeError(f"Pipeline输出 {output_name} 未生成")
            results[output_name] = context.get(output_name)
        
        print(f"\n{'='*60}")
        print(f"测试执行完成")
        print(f"{'='*60}\n")
        
        return results
```

def test_graph_add_node():
    """测试添加节点"""
    graph = TestGraph(name="test")

    """测试自动构建边"""
    graph = TestGraph(name="test")

    graph.add_node(node_b)
    graph.build_edges()
    
    # 验证边: x -> b
    assert "x" in graph.edges
    assert "b" in graph.edges["x"]


def test_graph_topological_sort():
    """测试拓扑排序"""
    graph = TestGraph(name="test")
    
    # A -> B -> C
    node_a = create_node("a", outputs=["x"])
    node_b = create_node("b", inputs=["x"], outputs=["y"])
    node_c = create_node("c", inputs=["y"])

    graph.add_node(node_c)

## 6. 使用示例

```python
from tge.core import TestGraph, TestEngine
from tge.ops import EnvCheckOp, ATCCompileOp

# 1. 创建测试图
graph = TestGraph(name="ATC_E2E_Test")

# 2. 添加节点
env_check = EnvCheckOp(name="env_check", attributes={"timeout": 60})
env_check.outputs = ["env_ready"]

atc_compile = ATCCompileOp(name="atc_compile", attributes={"timeout": 600})
atc_compile.inputs = ["env_ready", "model_path"]
atc_compile.outputs = ["om_path"]

graph.add_node(env_check)
graph.add_node(atc_compile)

# 3. 构建边
graph.build_edges()

# 4. 设置Pipeline输入输出
graph.pipeline_inputs = ["model_path"]
graph.pipeline_outputs = ["om_path"]

# 5. 执行
engine = TestEngine(graph)
result = engine.execute(inputs={"model_path": "/path/to/model.onnx"})

print(f"OM路径: {result['om_path']}")
```
