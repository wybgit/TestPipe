## 1. 需求描述

### 1.1 业务需求

TestPipe框架需要一个统一的测试算子基础框架,类似于AI编译器中的算子系统:
- 所有测试步骤抽象为测试算子(TestOp)
- 测试算子可注册、可发现、可复用
- 测试算子通过统一API访问底层能力
- 测试算子遵循统一模板,便于开发和维护

### 1.2 用户场景

**场景1: 开发新的测试算子**
```python
# 用户只需继承TestOp基类,实现execute方法
class MyTestOp(TestOp):
    op_type = "MyTest"
    
    def execute(self, context):
        # 使用统一API
        result = self.api.ssh.execute("ls -la")
        return {"output": result}
```

**场景2: 注册和使用测试算子**
```python
# 自动注册
@register_op
class MyTestOp(TestOp):
    pass

# 框架自动发现和调用
op = OpRegistry.get("MyTest")
```

## 2. 设计方案

### 2.1 核心类设计

#### TestOp基类

```python
from abc import ABC, abstractmethod
from typing import Dict, Any, List, Optional

class TestOp(ABC):
    """
    测试算子基类
    
    所有测试算子必须继承此类并实现execute方法
    """
    
    # ============================================
    # 类属性 - 算子元信息
    # ============================================
    op_type: str = ""           # 算子类型,如"ATCCompile"
    op_category: str = ""       # 算子分类,如"compile"
    op_version: str = "1.0"     # 算子版本
    
    # ============================================
    # 实例属性
    # ============================================
    def __init__(self, name: str, attributes: Dict[str, Any]):
        """
        初始化测试算子
        
        Args:
            name: 算子实例名称,如"atc_compile_1"
            attributes: 算子属性配置
        """
        self.name = name
        self.attributes = attributes
        self.inputs: List[str] = []      # 输入名称列表
        self.outputs: List[str] = []     # 输出名称列表
        self.api: OpAPI = None           # 统一操作API(由框架注入)
        self.logger = None               # 日志器(由框架注入)
    
    # ============================================
    # 核心方法 - 必须实现
    # ============================================
    @abstractmethod
    def execute(self, context: 'ExecutionContext') -> Dict[str, Any]:
        """
        执行算子逻辑
        
        Args:
            context: 执行上下文,包含:
                - 输入数据: context.get(input_name)
                - 工作目录: context.work_dir
                - 共享状态: context.metadata
        
        Returns:
            输出数据字典 {output_name: value}
            
        Raises:
            OpExecutionError: 执行失败时抛出
        """
        pass
    
    # ============================================
    # 可选方法 - 子类可覆盖
    # ============================================
    def validate(self) -> bool:
        """
        验证算子配置是否正确
        
        在execute之前调用,用于检查:
        - 必需的attributes是否存在
        - 参数值是否合法
        - 依赖的工具是否可用
        
        Returns:
            True表示验证通过,False表示验证失败
        """
        return True
    
    def setup(self, context: 'ExecutionContext'):
        """
        算子执行前的准备工作
        
        在execute之前调用,用于:
        - 创建临时目录
        - 初始化连接
        - 加载配置文件
        """
        pass
    
    def teardown(self, context: 'ExecutionContext'):
        """
        算子执行后的清理工作
        
        在execute之后调用(无论成功失败),用于:
        - 清理临时文件
        - 关闭连接
        - 释放资源
        """
        pass
    
    # ============================================
    # 辅助方法
    # ============================================
    def get_timeout(self) -> int:
        """获取超时时间(秒)"""
        return self.attributes.get('timeout', 300)
    
    def get_attribute(self, key: str, default: Any = None) -> Any:
        """获取属性值"""
        return self.attributes.get(key, default)
    
    def __repr__(self) -> str:
        return f"<{self.op_type}(name={self.name})>"
```

### 2.2 算子注册机制

#### OpRegistry - 算子注册表

```python
from typing import Dict, Type, List

class OpRegistry:
    """
    算子注册表
    
    管理所有可用的测试算子类型
    """
    
    _registry: Dict[str, Type[TestOp]] = {}
    
    @classmethod
    def register(cls, op_class: Type[TestOp]):
        """
        注册算子类
        
        Args:
            op_class: 算子类
        """
        op_type = op_class.op_type
        if not op_type:
            raise ValueError(f"算子类 {op_class.__name__} 必须定义 op_type")
        
        if op_type in cls._registry:
            raise ValueError(f"算子类型 {op_type} 已被注册")
        
        cls._registry[op_type] = op_class
        print(f"✓ 注册算子: {op_type} ({op_class.__name__})")
    
    @classmethod
    def get(cls, op_type: str) -> Type[TestOp]:
        """
        获取算子类
        
        Args:
            op_type: 算子类型
            
        Returns:
            算子类
            
        Raises:
            KeyError: 算子类型不存在
        """
        if op_type not in cls._registry:
            raise KeyError(f"未找到算子类型: {op_type}")
        
        return cls._registry[op_type]
    
    @classmethod
    def list_ops(cls) -> List[str]:
        """列出所有已注册的算子类型"""
        return list(cls._registry.keys())
    
    @classmethod
    def get_op_info(cls, op_type: str) -> Dict[str, Any]:
        """
        获取算子信息
        
        Returns:
            {
                'op_type': str,
                'op_category': str,
                'op_version': str,
                'class_name': str
            }
        """
        op_class = cls.get(op_type)
        return {
            'op_type': op_class.op_type,
            'op_category': op_class.op_category,
            'op_version': op_class.op_version,
            'class_name': op_class.__name__
        }


# 装饰器: 自动注册算子
def register_op(op_class: Type[TestOp]) -> Type[TestOp]:
    """
    装饰器: 自动注册算子
    
    用法:
        @register_op
        class MyTestOp(TestOp):
            op_type = "MyTest"
    """
    OpRegistry.register(op_class)
    return op_class
```

### 2.4 异常处理

```python
class OpError(Exception):
    """算子错误基类"""
    pass


class OpExecutionError(OpError):
    """算子执行错误"""
    
    def __init__(self, op_name: str, message: str, cause: Exception = None):
        self.op_name = op_name
        self.message = message
        self.cause = cause
        super().__init__(f"算子 {op_name} 执行失败: {message}")


class OpValidationError(OpError):
    """算子验证错误"""
    
    def __init__(self, op_name: str, message: str):
        self.op_name = op_name
        self.message = message
        super().__init__(f"算子 {op_name} 验证失败: {message}")


class OpTimeoutError(OpError):
    """算子超时错误"""
    
    def __init__(self, op_name: str, timeout: int):
        self.op_name = op_name
        self.timeout = timeout
        super().__init__(f"算子 {op_name} 执行超时 ({timeout}秒)")
```
